  function formatDate(v, blank) {
    if (!v) return blank === undefined ? '—' : blank;

    const d = m ? new Date(+m[1], +m[2] - 1, +m[3]) : new Date(v);
    if (isNaN(d.getTime())) return escHtml(v);
    return `${String(d.getDate()).padStart(2,'0')}/${String(d.getMonth()+1).padStart(2,'0')}/${d.getFullYear()}`;
  }

  async function apiPost(body) {
    if (!API_URL) {                       // โหมดตัวอย่าง: แก้ข้อมูลในหน้าเว็บอย่างเดียว
      if (body.action === 'add')    tableData.push({ ...body });
      if (body.action === 'edit')   tableData[body.rowIndex] = { ...tableData[body.rowIndex], ...body };
      if (body.action === 'delete') tableData[body.rowIndex].deleted = true;
      return;
    }
    const res  = await fetch(API_URL, { method:'POST', headers:{'Content-Type':'text/plain;charset=utf-8'}, body: JSON.stringify({ ...body, sheet: SHEET }) });
    const json = await res.json();
    if (json.status !== 'ok') throw new Error(json.message || 'error');
  }

  // ── Table ─────────────────────────────────────────────────────────────────
  function renderTable() {
    const q = searchTerm.toLowerCase();
    let rows = tableData.map((r, origIdx) => ({ ...r, origIdx })).filter(r => !r.deleted)
      .filter(r => !typeFilter || r.type === typeFilter)
      .filter(r => !q || [r.type, r.serial, r.operator, r.receiver, r.sender, r.condition, r.capacity].some(v => String(v || '').toLowerCase().includes(q)));

    if (sortKey) {
      rows.sort((a, b) => {
        let va = a[sortKey] || '', vb = b[sortKey] || '';
        if (sortKey === 'date' || sortKey === 'sentDate') {
          const ta = va ? new Date(va).getTime() : NaN, tb = vb ? new Date(vb).getTime() : NaN;
          if (isNaN(ta) && isNaN(tb)) return 0;
          if (isNaN(ta)) return 1;
          if (isNaN(tb)) return -1;
          return sortDir === 'asc' ? ta - tb : tb - ta;
        }
        va = String(va).toLowerCase(); vb = String(vb).toLowerCase();
        return (va < vb ? -1 : va > vb ? 1 : 0) * (sortDir === 'asc' ? 1 : -1);
      });
    }

    visibleRows = rows;
    renderNumHeader();

    if (!rows.length) { showMsg('📭', typeFilter ? `ไม่พบข้อมูลของ ${escHtml(typeFilter)}` : 'ไม่พบข้อมูล'); updateBulkBar(); return; }

    document.getElementById('tableBody').innerHTML = rows.map((r, i) => {
      const selectable = !r.sentDate;                 // ส่งแล้วจะเลือกซ้ำไม่ได้ในโหมดนี้
      const checked = bulkSel.has(r.origIdx);
      const firstCell = bulkMode
        ? `<input type="checkbox" class="bulk-cb" ${selectable ? '' : 'disabled'} ${checked && selectable ? 'checked' : ''} title="${selectable ? 'เลือกเครื่องนี้' : 'ส่งไปแล้ว'}"/>`
        : String(i + 1);
      const trCls = bulkMode ? (selectable ? 'bulk-row' : 'bulk-off') + (checked && selectable ? ' sel' : '') : '';
      return `<tr data-idx="${r.origIdx}" class="${trCls}">
      <td data-label="${bulkMode ? 'เลือก' : 'ลำดับ'}">${firstCell}</td>
      <td data-label="วันที่รับ">${formatDate(r.date)}</td>

  function parseCap(s) {

      };

    const ERR_T = 'บันทึกไม่สำเร็จ';
    if (!items.length) { showError(ERR_T, 'กรุณากรอกรหัส / S/N อย่างน้อย 1 เครื่อง'); return; }
    for (let i = 0; i < items.length; i++) {
      const x = items[i];
      if (!x.type || !x.serial) { showError(ERR_T, `รายการที่ ${i + 1}: กรุณาเลือกประเภทและกรอกรหัส / S/N ให้ครบ`); return; }
      if (!x.condition) { showError(ERR_T, `รายการที่ ${i + 1}: กรุณาเลือกลักษณะอุปกรณ์ (ใหม่ / เก่า)`); return; }
      if (x.type === HDD_TYPE && !(parseFloat(x.capNum) > 0)) { showError(ERR_T, `รายการที่ ${i + 1}: Harddisk ต้องกรอกความจุ`); return; }
    }

    // บังคับกรอกชื่อ: เพิ่มใหม่ต้องมี "ผู้รับเครื่อง" · ถ้ามีวันที่ส่ง (แก้ไข) ต้องมี "คนส่ง"
    const operator = document.getElementById('f-operator').value.trim();
    const receiver = document.getElementById('f-receiver').value.trim();
    const senderName = document.getElementById('f-senderEdit').value.trim();
    if (!isEdit && !receiver) {
      showError(ERR_T, 'กรุณากรอกชื่อผู้รับเครื่อง');
      document.getElementById('f-receiver').focus();
      return;
    }
    if (isEdit && document.getElementById('f-sentdate').value && !senderName) {
      showError('ส่งอุปกรณ์ไม่สำเร็จ', 'มีวันที่ส่งแล้ว กรุณากรอกชื่อคนส่งด้วย');
      document.getElementById('f-senderEdit').focus();
      return;
    }

    const seen = new Set();
    for (const x of items) {
      const k = (x.type + '|' + x.serial).toLowerCase();
      if (seen.has(k)) { showError(ERR_T, `รหัส ${x.serial} (${x.type}) ซ้ำกันในรายการที่กรอก`); return; }
      seen.add(k);
    }
    if (!isEdit) {
      const dups = items.filter(x => tableData.some(r => !r.deleted && r.type === x.type && String(r.serial).toLowerCase() === x.serial.toLowerCase()));
      if (dups.length && !confirm(`มีรหัสเหล่านี้ในระบบอยู่แล้ว:\n${dups.map(d => d.serial).join(', ')}\n\nต้องการเพิ่มซ้ำหรือไม่?`)) return;
    }

    const common = {
      sentDate: document.getElementById('f-sentdate').value || '',
      status:   document.getElementById('f-status').value || '—',
      operator, receiver,
      sender: senderName
    };
    const note = document.getElementById('f-note').value.trim();

    const btn = saveBtnEl;
    btn.disabled = true;
    let done = 0;
    try {
      if (isEdit) {
        btn.textContent = '⏳ กำลังบันทึก...';
        await apiPost({ action: 'edit', rowIndex: editingIdx, date: tableData[editingIdx].date, type: items[0].type, condition: items[0].condition, capacity: items[0].capacity, serial: items[0].serial, ...common });
      } else {
        const now = new Date().toISOString();
        for (const it of items) {
          btn.textContent = `⏳ กำลังบันทึก ${done + 1}/${items.length}`;
          await apiPost({ action: 'add', date: now, type: it.type, condition: it.condition, capacity: it.capacity, serial: it.serial, ...common });
          done++;
        }
      }
      closeModalFn();
      await loadData();
      if (!isEdit) {
        showSummary('add', items, receiver, new Date().toISOString(), note);
      } else {
        showSummary('edit',
          [{ type: items[0].type, capacity: items[0].capacity, serial: items[0].serial }],
          receiver, common.sentDate, '',
          { condition: items[0].condition, status: common.status, operator, receiver, sender: senderName, sentDate: common.sentDate });
      }
    } catch (err) {
      if (done > 0) {
        items.slice(0, done).forEach(x => x.el.remove());
        showError(ERR_T, `บันทึกสำเร็จ ${done} จาก ${items.length} เครื่อง แล้วเกิดข้อผิดพลาด: ${err.message}\nแถวที่เหลือยังอยู่ในฟอร์ม กดบันทึกอีกครั้งเพื่อลองใหม่`);

      }
    } finally { btn.disabled = false; updateDevUi(); }

  // ── Popup สรุป + ปุ่มคัดลอก ───────────────────────────────────────────────
  //  mode: 'add' = สรุปการรับ · 'send' = สรุปการส่ง · 'edit' = สรุปการแก้ไข
  function buildSummary(mode, items, person, dateIso, note, extra) {
    const isSend = mode === 'send', isEdit = mode === 'edit';
    // จัดกลุ่มตามชนิด (+ความจุ) ให้ S/N ของชนิดเดียวกันอยู่ด้วยกัน
    const groups = [];
    items.forEach(x => {
      const name = x.type + (x.capacity ? ` ${x.capacity}` : '');
      let g = groups.find(k => k.name === name);
      if (!g) { g = { name, serials: [] }; groups.push(g); }
      g.serials.push(x.serial);
    });
    const dateTxt = formatDate(dateIso).replace(/&[a-z#0-9]+;/g, '');
    const lines = [isEdit ? '✏️ สรุปการแก้ไขอุปกรณ์เสริม' : isSend ? '📦 สรุปการส่งอุปกรณ์เสริม' : '📦 สรุปการรับอุปกรณ์เสริม', ''];
    groups.forEach((g, i) => {
      lines.push(`${i + 1}. ${g.name}`);                 // เลขข้อกับชื่อชนิดอยู่บรรทัดเดียวกัน ไม่เว้นบรรทัด
      g.serials.forEach(sn => lines.push(`SN: ${sn}`));
      if (isEdit) {
        const e = extra || {};
        if (e.condition) lines.push(`ลักษณะ: ${e.condition}`);
        lines.push(`สถานะ: ${e.status || '-'}`);
        lines.push(`ผู้ดำเนินการ: ${e.operator || '-'}`);
        lines.push(`ผู้รับ: ${e.receiver || '-'}`);
        if (e.sentDate) {
          lines.push(`ผู้ส่ง: ${e.sender || '-'}`);
          lines.push(`วันที่ส่ง: ${formatDate(e.sentDate).replace(/&[a-z#0-9]+;/g, '')}`);
        }
      } else {
        lines.push(`${isSend ? 'ผู้ส่ง' : 'ผู้รับ'}: ${person}`);
        lines.push(`${isSend ? 'วันที่ส่ง' : 'วันที่รับ'}: ${dateTxt}`);
        lines.push(`หมายเหตุ: ${note || '-'}`);
      }
      lines.push('');
    });
    return lines.join('\n').trim();
  }

  //  (ใช้ได้เท่าที่กล้อง+เบราว์เซอร์รองรับ — Chrome บน Android รองรับครบ, iPhone Safari รองรับน้อยกว่า)

  function getTrack() {

  async function applyAdv(obj) {
    const t = getTrack();

  function initScanTools() {
    zoomRow.style.display = focusRow.style.display = torchRow.style.display = 'none';
    scanTools.style.display = 'none';
    torchOn = false; torchBtn.textContent = 'เปิด'; torchBtn.classList.remove('on');

      zoomSlider.min = scanCaps.zoom.min; zoomSlider.max = scanCaps.zoom.max; zoomSlider.step = scanCaps.zoom.step || 0.1;

      zoomRow.style.display = 'flex'; any = true;

    if (scanCaps.focusDistance && modes.includes('manual')) {
      focusSlider.min = scanCaps.focusDistance.min; focusSlider.max = scanCaps.focusDistance.max; focusSlider.step = scanCaps.focusDistance.step || 0.01;

      focusRow.style.display = 'flex'; any = true;

    focusRing.classList.toggle('fail', !ok);

  async function tapFocus(e) {
    const video = document.querySelector('#scanReader video');
    if (!video || !getTrack()) return;
    const wr = scanWrap.getBoundingClientRect(), vr = video.getBoundingClientRect();
    const nx = Math.min(1, Math.max(0, (e.clientX - vr.left) / vr.width));
    const ny = Math.min(1, Math.max(0, (e.clientY - vr.top) / vr.height));
    const modes = scanCaps.focusMode || [];
    let ok = false;
    if (modes.includes('single-shot') || modes.includes('continuous')) {
      const adv = { focusMode: modes.includes('single-shot') ? 'single-shot' : 'continuous' };
      if (scanCaps.pointsOfInterest) adv.pointsOfInterest = [{ x: nx, y: ny }];
      ok = await applyAdv(adv);
      if (!ok && adv.pointsOfInterest) { delete adv.pointsOfInterest; ok = await applyAdv(adv); }
      if (ok && adv.focusMode === 'single-shot' && modes.includes('continuous')) {
        clearTimeout(focusTimer);
        focusTimer = setTimeout(() => applyAdv({ focusMode: 'continuous' }), 1800);
      }
    }
    showRing(e.clientX - wr.left, e.clientY - wr.top, ok);
    if (ok) setScanMsg('🎯 กำลังโฟกัส... ถือนิ่งๆ สักครู่');
    else setScanMsg('เครื่องนี้ไม่รองรับสั่งโฟกัสจากหน้าเว็บ — ลองถือให้ห่างจากสติกเกอร์ประมาณ 15–25 ซม. แล้วรอให้ภาพชัด', true);
  }

  async function openScan(row) {
    if (scanStarting || scanner) return;
    scanRow = row; scanCount = 0; lastCode = ''; lastAt = 0;
    if (typeof showResults === 'function') showResults([]);
    scanOverlay.classList.add('active');
    if (typeof Html5Qrcode === 'undefined') { setScanMsg('โหลดตัวสแกนไม่สำเร็จ (ตรวจอินเทอร์เน็ต) — พิมพ์ S/N เองได้', true); return; }
    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
      setScanMsg('เปิดกล้องไม่ได้ — หน้านี้ต้องเปิดผ่าน https:// และเบราว์เซอร์ต้องรองรับกล้อง', true); return;
    }
    setScanMsg('กำลังเปิดกล้อง...');
    scanStarting = true;
    const F = window.Html5QrcodeSupportedFormats;
    const inst = new Html5Qrcode('scanReader', {
      verbose: false,
      formatsToSupport: [F.CODE_128, F.CODE_39, F.CODE_93, F.EAN_13, F.EAN_8, F.UPC_A, F.UPC_E, F.ITF, F.CODABAR, F.QR_CODE, F.DATA_MATRIX],
      experimentalFeatures: { useBarCodeDetectorIfSupported: true }
    });
    scanner = inst;
    const baseCfg = { fps: 12, aspectRatio: 1.333 };   // ไม่กำหนด qrbox = สแกนทั้งเฟรม (บาร์โค้ดเล็กบนสติกเกอร์จะไม่หลุดกรอบ)
    try {
      try {
        // ขอภาพความละเอียดสูง (บาร์โค้ดเล็กต้องใช้ภาพละเอียด) + โฟกัสต่อเนื่อง
        await inst.start(
          { facingMode: 'environment' },
          { ...baseCfg, videoConstraints: {
              facingMode: { ideal: 'environment' },
              width: { ideal: 1920 }, height: { ideal: 1080 },
              advanced: [{ focusMode: 'continuous' }]
          } },
          onScanOk, () => {}
        );
      } catch (e1) {
        if (/NotAllowed|Permission|NotFound|NotReadable/i.test((e1 && e1.name) || String(e1))) throw e1;
        await inst.start({ facingMode: 'environment' }, baseCfg, onScanOk, () => {});
      }
      if (!scanOverlay.classList.contains('active')) { try { await inst.stop(); } catch (e) {} return; }
      initScanTools();
      setScanMsg('แตะที่ภาพเพื่อโฟกัส · ถ้าสแกนสดไม่ติด กด "จับภาพนี้แล้วสแกน"');
    } catch (err) {
      if (scanner === inst) scanner = null;
      const n = (err && err.name) || String(err);
      setScanMsg(/NotAllowed|Permission/i.test(n) ? 'ไม่ได้อนุญาตให้ใช้กล้อง — กดไอคอนกุญแจข้างช่อง URL แล้วอนุญาตกล้อง'
               : /NotFound|Devices/i.test(n)      ? 'ไม่พบกล้องในอุปกรณ์นี้'
               : /NotReadable|Abort/i.test(n)     ? 'กล้องถูกใช้งานโดยแอปอื่นอยู่'
               : 'เปิดกล้องไม่สำเร็จ: ' + n, true);
    } finally { scanStarting = false; }
  }

  function onScanOk(text) {
    const code = String(text || '').trim();
    const now = Date.now();
    if (!code || !scanRow || !scanRow.isConnected) return;
    if (code === lastCode && now - lastAt < 2000) return;
    lastCode = code; lastAt = now;
    const inp = scanRow.querySelector('.dev-serial');
    const dup = [...devList.querySelectorAll('.dev-serial')].some(i => i !== inp && i.value.trim().toLowerCase() === code.toLowerCase());
    if (dup) { setScanMsg(`⚠️ ${code} ซ้ำกับที่สแกนไว้แล้ว`, true); return; }
    inp.value = code;
    scanCount++;
    if (navigator.vibrate) navigator.vibrate(60);
    updateDevUi();
    if (document.getElementById('scanContinuous').checked && editingIdx === null) {
      scanRow = addDevRowLike(scanRow, false);
      scanRow.scrollIntoView({ block: 'nearest' });
      setScanMsg(`✅ ${code} · สแกนแล้ว ${scanCount} เครื่อง — สแกนเครื่องถัดไปได้เลย`);
    } else {
      closeScan(); inp.focus();
    }
  }

  // ── ถอดรหัสจากภาพนิ่ง (ความละเอียดเต็ม) ──────────────────────────────────
  //  1) BarcodeDetector ของเบราว์เซอร์ (ถ้ามี)  2) ZXing แบบซอยภาพเป็นช่วงๆ + ขยาย เพื่อจับบาร์โค้ดเล็กบนสติกเกอร์

  const tick = () => new Promise(r => setTimeout(r, 0));

  function makeCanvas(w, h) { const c = document.createElement('canvas'); c.width = Math.max(1, Math.round(w)); c.height = Math.max(1, Math.round(h)); return c; }
  function fitCanvas(src, sw, sh, maxSide) {          // ย่อถ้าใหญ่เกิน (กันเครื่องค้าง) แต่ไม่ย่อจนเล็ก
    const k = Math.min(1, maxSide / Math.max(sw, sh));
    const c = makeCanvas(sw * k, sh * k);
    c.getContext('2d').drawImage(src, 0, 0, c.width, c.height);
    return c;
  }
  async function decodeCanvas(base) {
    const found = new Map();
    const add = (t, f) => { t = String(t || '').trim(); if (t && !found.has(t)) found.set(t, f || ''); };

    if ('BarcodeDetector' in window) {
      try { (await new BarcodeDetector().detect(base)).forEach(r => add(r.rawValue, r.format)); } catch (e) {}
    }
    if (typeof ZXing !== 'undefined') {
      const Z = ZXing, hints = new Map();
      hints.set(Z.DecodeHintType.TRY_HARDER, true);
      hints.set(Z.DecodeHintType.POSSIBLE_FORMATS, [Z.BarcodeFormat.CODE_128, Z.BarcodeFormat.CODE_39, Z.BarcodeFormat.CODE_93,
        Z.BarcodeFormat.EAN_13, Z.BarcodeFormat.EAN_8, Z.BarcodeFormat.UPC_A, Z.BarcodeFormat.UPC_E, Z.BarcodeFormat.ITF,
        Z.BarcodeFormat.CODABAR, Z.BarcodeFormat.QR_CODE, Z.BarcodeFormat.DATA_MATRIX]);
      const reader = new Z.MultiFormatReader(); reader.setHints(hints);
      const tryCanvas = c => {
        for (const B of [Z.HybridBinarizer, Z.GlobalHistogramBinarizer]) {
          try {
            const bm = new Z.BinaryBitmap(new B(new Z.HTMLCanvasElementLuminanceSource(c)));
            const r = reader.decode(bm);
            add(r.getText(), Z.BarcodeFormat[r.getBarcodeFormat()]);
            return;
          } catch (e) {}
        }
      };
      const W = base.width, H = base.height;
      for (const n of [1, 2, 3, 4]) {                    // ทั้งภาพ → ซอย 2×2 → 3×3 → 4×4 (ช่องซ้อนกัน)
        const ww = n === 1 ? W : Math.min(W, (W / n) * 1.6), hh = n === 1 ? H : Math.min(H, (H / n) * 1.6);
        for (let iy = 0; iy < n; iy++) for (let ix = 0; ix < n; ix++) {
          const x = n === 1 ? 0 : (W - ww) * ix / (n - 1), y = n === 1 ? 0 : (H - hh) * iy / (n - 1);
          const k = Math.min(3, Math.max(1, 1000 / ww));   // ขยายช่องเล็กให้บาร์โค้ดหนาพอ
          const c = makeCanvas(ww * k, hh * k);
          const g = c.getContext('2d'); g.imageSmoothingEnabled = true;
          g.drawImage(base, x, y, ww, hh, 0, 0, c.width, c.height);
          tryCanvas(c);
          await tick();
        }
      }
    }
    return [...found].map(([text, format]) => ({ text, format }));
  }

  function showResults(list) {
    scanResults.innerHTML = '';
    scanResults.style.display = 'none';
    if (!list.length) return;
    scanResults.style.display = 'flex';
    const t = document.createElement('div'); t.className = 'res-title'; t.textContent = `พบ ${list.length} รหัส — แตะอันที่เป็น S/N`;
    scanResults.appendChild(t);
    list.forEach(r => {
      const b = document.createElement('button'); b.type = 'button'; b.className = 'code-chip';
      b.innerHTML = `<span>${escHtml(r.text)}</span><small>${escHtml(String(r.format).replace(/_/g, ' '))}</small>`;
      b.addEventListener('click', () => { lastCode = ''; onScanOk(r.text); if (scanOverlay.classList.contains('active')) scanResults.style.display = 'none'; });
      scanResults.appendChild(b);
    });
  }

  async function runDecode(canvas) {
    setScanMsg('⏳ กำลังอ่านบาร์โค้ดจากภาพ...');
    grabBtn.disabled = photoBtn.disabled = true;
    try {
      const list = await decodeCanvas(canvas);
      if (!scanOverlay.classList.contains('active')) return;
      if (!list.length) {
        showResults([]);
        setScanMsg('อ่านไม่ได้ — ถือห่างสติกเกอร์ ~15–25 ซม. ให้บาร์โค้ดชัดและไม่สะท้อนแสง (เปิดไฟฉาย/หลบแสงสะท้อน) แล้วลองใหม่', true);
      } else if (list.length === 1) {
        showResults([]);
        lastCode = ''; onScanOk(list[0].text);
      } else {
        setScanMsg('พบหลายบาร์โค้ดบนสติกเกอร์ — เลือกอันที่เป็น S/N');
        showResults(list);
      }
    } catch (err) {
      setScanMsg('อ่านภาพไม่สำเร็จ: ' + (err && err.message || err), true);
    } finally { grabBtn.disabled = photoBtn.disabled = false; }
  }

  // จับภาพจากกล้องที่เปิดอยู่ (พยายามใช้ภาพถ่ายความละเอียดเต็มของเซนเซอร์ก่อน)
  grabBtn.addEventListener('click', async () => {
    const video = document.querySelector('#scanReader video');
    if (!video || !video.videoWidth) { setScanMsg('กล้องยังไม่พร้อม', true); return; }
    showResults([]);
    setScanMsg('📸 กำลังจับภาพ... ถือนิ่งๆ');
    let canvas = null;
    const track = getTrack();
    if (track && 'ImageCapture' in window) {
      try {
        const blob = await Promise.race([new ImageCapture(track).takePhoto(), new Promise((_, rej) => setTimeout(() => rej(new Error('timeout')), 4000))]);
        const bmp = await createImageBitmap(blob);
        canvas = fitCanvas(bmp, bmp.width, bmp.height, 2600);
      } catch (e) { canvas = null; }
    }
    if (!canvas) canvas = fitCanvas(video, video.videoWidth, video.videoHeight, 2600);
    runDecode(canvas);
  });

    } catch (e) { setScanMsg('เปิดรูปไม่ได้: ' + (e && e.message || e), true); }
  });

  document.getElementById('closeScanBtn').addEventListener('click', closeScan);
